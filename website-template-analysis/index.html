Khám phá chủ đề Xem lộ trình

Nội dung dễ tiếp cận cho người mới bắt đầu.

hello.py 
# Chương trình đầu tiên
def greet(name):
    return f"Xin chào, {name}!"

message = greet("bạn")
print(message)
